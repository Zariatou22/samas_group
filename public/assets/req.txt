SELECT kl.*, ka.auth_number AS auth_number, ka.quantity AS auth_quantity, p.name AS product_name,p.container_type AS container_type, kc.full_registration AS car_registration, kc.capability AS capability, ks.name AS source_name, kcc.name AS customer_name, s.name AS ship_name, s.arrival_date AS ship_date, sp.ship_products_id AS ship_product, sp.bl AS bl, sp.quantity AS bl_quantity, ko.id AS driver, ko.name AS owner_name, ko.contact AS owner_contact, kd.id AS driver, kd.name AS driver_name,kd.contact AS driver_contact, kb.id AS id FROM kanis_loading kl INNER JOIN ship s ON kl.ship=s.ship_id INNER JOIN kanis_authorizations ka ON kl.authorization=ka.id INNER JOIN products p ON kl.product=p.id INNER JOIN kanis_cars kc ON kl.car=kc.id INNER JOIN kanis_sources ks ON kl.source=ks.id INNER JOIN kanis_customers kcc ON kl.customer=kcc.id INNER JOIN ship_products sp ON kl.ship=sp.ship INNER JOIN kanis_owners ko ON kc.owner=ko.id INNER JOIN kanis_drivers kd ON kc.driver=kd.id LEFT JOIN kanis_box kb ON kl.id=kb.loading WHERE sp.product=kl.product AND sp.ship_products_id=ka.ship_product AND kc.full_registration LIKE :reg AND kb.id IS NULL
